    "use strict";

    // 1. Find the elements used by the generator.
    const heroPreview = document.getElementById(
      "hero-preview"
    );

    const heroHeading = document.getElementById(
      "hero-heading"
    );

    const heroSubheading = document.getElementById(
      "hero-subheading"

      "height-checkbox"
    );

    const subheadingCheckbox = document.getElementById(
      "subheading-checkbox"
    );

    const h1SerifCheckbox = document.getElementById(
      "h1-serif-checkbox"
    );

    const h2SerifCheckbox = document.getElementById(
      "h2-serif-checkbox"
    );

    const h1LetterSpacingSlider = document.getElementById(
      "h1-letter-spacing"
    );

    const h2LetterSpacingSlider = document.getElementById(
      "h2-letter-spacing"
    );

    const h1LetterSpacingValue = document.getElementById(
      "h1-letter-spacing-value"
    );

    const h2LetterSpacingValue = document.getElementById(
      "h2-letter-spacing-value"
    );

    const h2Effects = document.getElementById(
      "h2-effects"
    );

    const heroLetterSpacing = document.getElementById(
      "hero-letter-spacing"
    );

    const copyButton = document.getElementById(
      "copy-button"
    );

    const copyStatus = document.getElementById(
      "copy-status"
    );

    const fallbackPanel = document.getElementById(
      "fallback-panel"
    );

    const fallbackTextarea = document.getElementById(
      "fallback-html"
    );

    const heroStyles = document.getElementById(
      "hero-styles"
    ).textContent;

    // 2. Store the current settings, including text effects.
    const settings = {
      dark: false,
      centre: false,
      minimumHeight: true,
      showSubheading: true,
      h1Serif: false,
      h2Serif: false,
      h1LetterSpacing: -0.05,
      h2LetterSpacing: -0.025
    };

    // Read the actual state of each control.
    function readSettings() {
      settings.dark = darkCheckbox.checked;
      settings.centre = centreCheckbox.checked;
      settings.minimumHeight = heightCheckbox.checked;
      settings.showSubheading = subheadingCheckbox.checked;

      settings.h1Serif = h1SerifCheckbox.checked;
      settings.h2Serif = h2SerifCheckbox.checked;

      settings.h1LetterSpacing = Number(
        h1LetterSpacingSlider.value
      );

      settings.h2LetterSpacing = Number(
        h2LetterSpacingSlider.value
      );
    }

    // Format numeric letter spacing consistently in em.
    function formatLetterSpacing(value) {
      const normalised = Math.abs(value) < 0.0005
        ? 0
        : value;

      return normalised.toFixed(3) + "em";
    }

    // Build heading-specific CSS without inline styles.
    function getLetterSpacingCss() {
      return `.hero #hero-heading {
  letter-spacing: ${formatLetterSpacing(
    settings.h1LetterSpacing
  )};
}

.hero #hero-subheading {
  letter-spacing: ${formatLetterSpacing(
    settings.h2LetterSpacing
  )};
}`;
    }

    // 3. Build the hero's class list.
    // This same function is used for preview and export.
    function getHeroClasses() {
      const classes = ["hero"];

      classes.push(
        settings.dark ? "hero-dark" : "hero-light"
      );

      if (settings.centre) {
        classes.push("hero-centre");
      }

      if (settings.minimumHeight) {
        classes.push("hero-min-height");
      }

      return classes.join(" ");
    }

    // Build the font class for each heading.
    function getHeadingClasses(headingLevel) {
      if (headingLevel === "h1") {
        return settings.h1Serif
          ? "hero-h1-serif"
          : "hero-h1-sans";
      }

      if (headingLevel === "h2") {
        return settings.h2Serif
          ? "hero-h2-serif"
          : "hero-h2-sans";
      }

      return "";
    }

    // 4. Update classes without replacing editable headings.
    function updatePreview() {
      heroPreview.className = getHeroClasses();

      heroHeading.className = getHeadingClasses("h1");
      heroSubheading.className = getHeadingClasses("h2");

      heroSubheading.hidden = !settings.showSubheading;

      // Disable H2 controls without changing their values.
      h2Effects.disabled = !settings.showSubheading;

      // Update both displayed slider values.
      h1LetterSpacingValue.textContent =
        formatLetterSpacing(settings.h1LetterSpacing);

      h2LetterSpacingValue.textContent =
        formatLetterSpacing(settings.h2LetterSpacing);

      // Update the dedicated head style element.
      // No inline styles are applied to the headings.
      heroLetterSpacing.textContent =
        getLetterSpacingCss();
    }

    // 5. Get the latest plain text from an editable heading.
    function getHeadingText(element) {
      return element.textContent;
    }

    // Escape text before inserting it into exported HTML.
    function escapeHtml(text) {
      return text
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#39;");
    }

    // Insert pasted content as plain text only.
    function handlePlainTextPaste(event) {
      event.preventDefault();

      const pastedText = (
        event.clipboardData &&
        event.clipboardData.getData("text/plain")
      ) || "";

      insertPlainText(pastedText);
    }

    // Insert text at the current caret without HTML formatting.
    function insertPlainText(text) {
      const selection = window.getSelection();

      if (!selection || selection.rangeCount === 0) {
        return;
      }

      const range = selection.getRangeAt(0);
      const activeElement = document.activeElement;

      if (
        activeElement !== heroHeading &&
        activeElement !== heroSubheading
      ) {
        return;
      }

      if (!activeElement.contains(range.commonAncestorContainer)) {
        return;
      }

      range.deleteContents();

      const textNode = document.createTextNode(
        text.replace(/[\r\n]+/g, " ")
      );

      range.insertNode(textNode);
      range.setStartAfter(textNode);
      range.collapse(true);

      selection.removeAllRanges();
      selection.addRange(range);

      // Notify the generator about the text change.
      activeElement.dispatchEvent(
        new Event("input", { bubbles: true })
      );
    }

    // Prevent Enter from inserting paragraphs or line breaks.
    function preventNewLines(event) {
      if (event.key === "Enter") {
        event.preventDefault();
      }
    }

    // Keep editable headings limited to plain text nodes.
    function normaliseHeading(element) {
      const selection = window.getSelection();
      const isFocused = document.activeElement === element;

      // Do not rebuild the heading during normal typing.
      // This preserves the caret position.
      if (isFocused) {
        return;
      }

      const plainText = element.textContent;

      if (
        element.childNodes.length !== 1 ||
        element.firstChild.nodeType !== Node.TEXT_NODE
      ) {
        element.textContent = plainText;
      }
    }

    // 6. Generate a complete standalone HTML document.
    function generateHtml() {
      const headingText = escapeHtml(
        getHeadingText(heroHeading)
      );

      const subheadingText = escapeHtml(
        getHeadingText(heroSubheading)
      );

      const heroClasses = getHeroClasses();

      const h1Classes = getHeadingClasses("h1");
      const h2Classes = getHeadingClasses("h2");

      const letterSpacingCss = getLetterSpacingCss();

      const subheadingHtml = settings.showSubheading
        ? `\n        <h2 id="hero-subheading" class="${h2Classes}">${subheadingText}</h2>`
        : "";

      // Escaped closing script tag prevents the browser from
      // ending this generator's own script prematurely.
      const tailwindScript =
        '<script src="https://cdn.jsdelivr.net/npm/@tailwindcss/browser@4"><\/script>';

      return `<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <meta name="description" content="Short, unique description of web page. Often used as the snippet shown in search results. About 120 characters is ideal.">
  <title>Hero Block</title>

  ${tailwindScript}

  <style type="text/tailwindcss">
${heroStyles.trim()}
  </style>

  <style id="hero-letter-spacing">
${letterSpacingCss}
  </style>
</head>
<body>
  <header class="${heroClasses}">
    <div class="hero-content">
      <h1 id="hero-heading" class="${h1Classes}">${headingText}</h1>${subheadingHtml}
    </div>
  </header>
</body>
</html>`;
    }

    // 7. Keep manual-copy HTML current when it is visible.
    function updateFallback() {
      if (!fallbackPanel.hidden) {
        fallbackTextarea.value = generateHtml();
      }
    }

    // Update the preview after a checkbox changes.
    function handleSettingsChange() {
      readSettings();
      updatePreview();
      updateFallback();
    }

    // Update letter spacing immediately while dragging.
    function handleLetterSpacingInput() {
      readSettings();
      updatePreview();
      updateFallback();
    }

    // Handle heading edits.
    function handleHeadingInput() {
      updateFallback();
    }

    // Show manual-copy HTML if the Clipboard API fails.
    function showManualCopy(html) {
      fallbackTextarea.value = html;
      fallbackPanel.hidden = false;

      copyStatus.textContent =
        "Automatic copying is unavailable. " +
        "Use the manual-copy box below.";
    }

    // 8. Copy the complete exported document.
    async function copyHtml() {
      const html = generateHtml();

      copyStatus.textContent = "";

      try {
        if (
          !navigator.clipboard ||
          typeof navigator.clipboard.writeText !== "function"
        ) {
          throw new Error("Clipboard API unavailable");
        }

        await navigator.clipboard.writeText(html);

        copyStatus.textContent =
          "HTML copied. Paste it into VS Code.";

      } catch (error) {
        showManualCopy(html);
      }
    }

    // 9. Connect the controls to their functions.
    const checkboxes = [
      darkCheckbox,
      centreCheckbox,
      heightCheckbox,
      subheadingCheckbox,
      h1SerifCheckbox,
      h2SerifCheckbox
    ];

    checkboxes.forEach(function (checkbox) {
      checkbox.addEventListener(
        "change",
        handleSettingsChange
      );
    });

    // Native range sliders update continuously on input.
    const letterSpacingSliders = [
      h1LetterSpacingSlider,
      h2LetterSpacingSlider
    ];

    letterSpacingSliders.forEach(function (slider) {
      slider.addEventListener(
        "input",
        handleLetterSpacingInput
      );
    });

    const editableHeadings = [
      heroHeading,
      heroSubheading
    ];

    editableHeadings.forEach(function (heading) {
      heading.addEventListener(
        "paste",
        handlePlainTextPaste
